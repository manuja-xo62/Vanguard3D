# 1. FIX: Pinned base image to a minimal, specific release version
FROM alpine:3.20

# 2. FIX: Set dedicated, non-root application directory
WORKDIR /app

# 3. FIX: Install minimal dependencies and clear package manager cache inline
RUN apk add --no-cache ca-certificates tzdata

# 4. FIX: Create a dedicated non-root user and group
RUN addgroup -S appgroup && adduser -S appuser -G appgroup

# 5. FIX: Bounded file copy directly into WORKDIR with correct user ownership
COPY --chown=appuser:appgroup . /app

# 6. FIX: Least-privilege file permissions (755 instead of 777)
RUN chmod -R 755 /app

# 7. FIX: Expose only non-privileged application port
EXPOSE 8080

# 8. FIX: Include health check instruction for orchestrator monitoring
HEALTHCHECK --interval=30s --timeout=3s --retries=3 \
  CMD wget --quiet --tries=1 --spider http://localhost:8080/ || exit 1

# 9. FIX: Switch explicitly to non-root execution context
USER appuser

# 10. FIX: Run standard application binary (avoiding SSH daemon or shell execution)
CMD ["node", "server.js"]
